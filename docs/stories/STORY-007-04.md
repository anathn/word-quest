## Overview

Display simple, encouraging progress statistics that are easy for 3rd graders to understand. Focus on positive metrics that celebrate achievements rather than highlighting failures.

## User Story

As a student, I want to see simple numbers about my progress so that I can see how well I'm doing and feel proud of my achievements.

## Acceptance Criteria

### Functional Requirements

- [ ] Display "Words Mastered" counter (e.g., "23/50 words")
- [ ] Display accuracy rate with encouraging format ("You get it right 8 out of 10 times!")
- [ ] Show best streak achievement ("Your best streak: 7!")
- [ ] Display total time practiced in kid-friendly format ("You've practiced for 2 hours!")
- [ ] Show today's practice summary ("Today: 8 words, 6 correct!")
- [ ] Include upward trend indicators for improving metrics
- [ ] Keep all numbers positive - no percentages below 50% shown as failures
- [ ] Refresh stats after each completed planet

### Non-Functional Requirements

- [ ] Stats load within 300ms
- [ ] Large, clear numbers (48pt+ for counters)
- [ ] High contrast display
- [ ] Update in real-time after each word completion
- [ ] Accessibility: Screen reader announces stats correctly

class StatsCalculator:
    def __init__(self, progress_data):
        self.progress = progress_data
    
    def get_words_mastered(self):
        """Return total words mastered count"""
        return len(self.progress.mastered_words)
    
    def get_total_words(self):
        """Return total words in current curriculum"""
        return len(self.progress.all_words)
    
    def get_accuracy_display(self):
        """Return accuracy in kid-friendly format"""
        total_attempts = self.progress.total_attempts
        correct_answers = self.progress.correct_answers
        
        if total_attempts == 0:
            return {"text": "Keep practicing!", "value": 0}
        
        accuracy = correct_answers / total_attempts
        
        # Convert to "X out of Y" format
        # Round to nearest 10 for simplicity
        denominator = 10
        numerator = round(accuracy * denominator)
        
        return {
            "text": f"You get it right {numerator} out of {denominator} times!",
            "value": accuracy,
            "trend": self.calculate_accuracy_trend()
        }
    
    def get_best_streak(self):
        """Return longest streak ever achieved"""
        return self.progress.best_streak
    
    def get_total_practice_time(self):
        """Return total time practiced in friendly format"""
        total_seconds = self.progress.total_practice_seconds
        
        if total_seconds < 300:  # Less than 5 minutes
            return "Just getting started!"
        elif total_seconds < 3600:  # Less than 1 hour
            minutes = total_seconds // 60
            return f"You've practiced for {minutes} minutes!"
        elif total_seconds < 7200:  # Less than 2 hours
            hours = total_seconds // 3600
            minutes = (total_seconds % 3600) // 60
            return f"You've practiced for {hours} hour and {minutes} minutes!"
        else:
            hours = total_seconds // 3600
            return f"You've practiced for {hours} hours!"
    
    def get_today_summary(self):
        """Return today's practice summary"""
        today = datetime.now().date()
        today_data = self.progress.get_day_data(today)
        
        if not today_data:
            return None
        
        correct = today_data.correct_count
        total = today_data.total_count
        
        return {
            "words": total,
            "correct": correct,
            "text": f"Today: {total} words, {correct} correct!" if correct > 0 else f"Today: {total} words!"
        }
    
    def calculate_accuracy_trend(self):
        """Calculate if accuracy is improving, stable, or declining"""
        recent = self.progress.get_week_accuracy()
        previous = self.progress.get_previous_week_accuracy()
        
        if previous == 0:
            return "new"
        
        diff = recent - previous
        
        if diff >= 0.05:
            return "up"  # Improving
        elif diff <= -0.05:
            return "down"  # Declining
        return "stable"

class ProgressStats:
    def __init__(self, screen, progress_data):
        self.screen = screen
        self.calculator = StatsCalculator(progress_data)
        self.stat_cards = []
        self.layout = self.calculate_layout()
        self.build_stat_cards()
    
    def build_stat_cards(self):
        """Create stat card components"""
        self.stat_cards = [
            StatCard(
                "Words Mastered",
                f"{self.calculator.get_words_mastered()}/{self.calculator.get_total_words()}",
                "⭐"
            ),
            StatCard(
                "Accuracy",
                self.calculator.get_accuracy_display()["text"],
                "🎯"
            ),
            StatCard(
                "Best Streak",
                f"{self.calculator.get_best_streak()} correct!",
                "🔥"
            ),
            StatCard(
                "Practice Time",
                self.calculator.get_total_practice_time(),
                "⏱️"
            ),
        ]
        
        today_summary = self.calculator.get_today_summary()
        if today_summary:
            self.stat_cards.append(
                StatCard("Today", today_summary["text"], "📅")
            )
    
    def calculate_layout(self):
        """Calculate card positions in 2x3 grid"""
        cards = 5  # or 6 with today
        cols = 3
        card_width = 280
        card_height = 120
        spacing = 20
        start_x = 50
        start_y = 100
        
        layout = []
        for i, card in enumerate(self.stat_cards):
            col = i % cols
            row = i // cols
            x = start_x + (col * (card_width + spacing))
            y = start_y + (row * (card_height + spacing))
            layout.append((x, y, card))
        
        return layout
    
    def update_stats(self):
        """Refresh all stats (call after progress changes)"""
        self.build_stat_cards()
    
    def draw(self):
        """Render all stat cards"""
        self.draw_background()
        
        for x, y, card in self.layout:
            card.draw(self.screen, x, y)
        
        self.draw_title("Your Progress")

class StatCard:
    def __init__(self, title, value, icon):
        self.title = title
        self.value = value
        self.icon = icon
        self.card_rect = None
    
    def draw(self, screen, x, y):
        """Draw stat card"""
        # Card background
        card_rect = pygame.Rect(x, y, 280, 120)
        self.card_rect = card_rect
        
        pygame.draw.rect(screen, DARK_BLUE, card_rect, border_radius=15)
        pygame.draw.rect(screen, GOLD, card_rect, width=2, border_radius=15)
        
        # Icon
        render_text(screen, self.icon, 40, x+15, y+15)
        
        # Title
        render_text(screen, self.title, 18, x+60, y+15, bold=True)
        
        # Value
        render_text(screen, self.value, 24, x+15, y+55, bold=True, color=WHITE)
