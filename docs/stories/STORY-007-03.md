## Overview

Create a progress journal that displays encouraging, age-appropriate text summaries of the student's journey. The journal uses positive language to celebrate achievements and provide constructive feedback.

## User Story

As a student, I want to read about my progress in simple, encouraging words so that I understand how I'm doing and feel motivated to keep practicing.

## Acceptance Criteria

### Functional Requirements

- [ ] Display weekly progress summary with encouraging language
- [ ] Show words mastered this week with list
- [ ] Show words needing practice with gentle suggestions
- [ ] Include positive observations about improvement
- [ ] Display total words mastered count
- [ ] Show streak achievements ("Your longest streak: 7!")
- [ ] Navigate between weekly journal entries
- [ ] Use 3rd grade reading level (simple words, short sentences)

### Non-Functional Requirements

- [ ] Text generation within 500ms
- [ ] Clear, readable typography (24pt+ minimum)
- [ ] High contrast text (white on dark background)
- [ ] Accessibility: Screen reader compatible
- [ ] Language tone consistently encouraging

```python
class ProgressSummarizer:
    def __init__(self, progress_data):
        self.progress = progress_data
        self.templates = LanguageTemplates()
    
    def generate_weekly_summary(self, week_start, week_end):
        """Generate encouraging summary for a week"""
        week_data = self.progress.get_week_data(week_start, week_end)
        
        summary_parts = []
        
        # Opening encouragement
        summary_parts.append(self.templates.get_opening(week_data.total_words))
        
        # Words mastered
        if week_data.mastered_words:
            mastered_list = ", ".join(week_data.mastered_words[:5])
            if len(week_data.mastered_words) > 5:
                mastered_list += " and more!"
            summary_parts.append(f"This week you mastered: {mastered_list}")
        
        # Progress observation
        if week_data.accuracy_improved:
            summary_parts.append("You're getting faster at spelling!")
        elif week_data.struggling_words:
            summary_parts.append("Keep practicing - you're making progress!")
        
        # Streak highlight
        if week_data.best_streak >= 5:
            summary_parts.append(f"Amazing! Your streak of {week_data.best_streak} correct was fantastic!")
        
        # Closing encouragement
        summary_parts.append(self.templates.get_closing(week_data.mastered_count))
        
        return {
            "period": f"Week of {week_start.strftime('%B %d')}",
            "lines": summary_parts,
            "mastered_count": len(week_data.mastered_words),
            "needs_practice": week_data.weak_words
        }
    
    def get_needs_practice_section(self, weak_words):
        """Generate gentle suggestion for practice words"""
        if not weak_words:
            return None
        
        word_list = ", ".join([w.name for w in weak_words[:3]])
        return {
            "title": "Words to Practice",
            "text": f"Keep practicing: {word_list}",
            "tone": "encouraging",
            "words": weak_words
        }

class LanguageTemplates:
    """Pre-written phrases at 3rd grade reading level"""
    
    OPENINGS = {
        "active": [
            "Wow, you're really working hard!",
            "Great job this week!",
            "You're doing awesome!",
            "Look how much you've practiced!"
        ],
        "steady": [
            "Good practice this week!",
            "You're making progress!",
            "Keep it up! Great work!",
            "You're getting better every day!"
        ],
        "light": [
            "Nice job this week!",
            "Good job practicing!",
            "You're doing great!",
            "Well done!"
        ]
    }
    
    CLOSEINGS = {
        "many_mastered": [
            "Keep going - you're a spelling star!",
            "You're getting so good at spelling!",
            "Amazing work! Keep it up!",
            "You're a spelling champion!"
        ],
        "some_mastered": [
            "Keep practicing and you'll get even better!",
            "You're on your way! Great work!",
            "Keep going - you can do it!",
            "Every word you learn makes you stronger!"
        ],
        "few_mastered": [
            "Keep trying - you're making progress!",
            "Learning takes time. You're doing great!",
            "Mistakes help us learn. Keep going!",
            "Practice makes progress. You've got this!"
        ]
    }
    
    def get_opening(self, word_count):
        """Select appropriate opening based on activity level"""
        if word_count >= 10:
            return random.choice(self.OPENINGS["active"])
        elif word_count >= 5:
            return random.choice(self.OPENINGS["steady"])
        else:
            return random.choice(self.OPENINGS["light"])
    
    def get_closing(self, mastered_count):
        """Select appropriate closing based on success"""
        if mastered_count >= 5:
            return random.choice(self.CLOSEINGS["many_mastered"])
        elif mastered_count >= 2:
            return random.choice(self.CLOSEINGS["some_mastered"])
        else:
            return random.choice(self.CLOSEINGS["few_mastered"])

class ProgressJournal:
    def __init__(self, screen, progress_data):
        self.screen = screen
        self.summarizer = ProgressSummarizer(progress_data)
        self.current_week = 0
        self.entries = self.load_entries()
    
    def load_entries(self):
        """Generate journal entries for recent weeks"""
        entries = []
        current = datetime.now()
        
        for i in range(4):  # Last 4 weeks
            week_start = current - timedelta(weeks=i+1)
            week_end = current - timedelta(weeks=i)
            
            entry = self.summarizer.generate_weekly_summary(week_start, week_end)
            entries.append(entry)
        
        return entries
    
    def handle_input(self, event):
        """Navigate between weeks"""
        if event.type == pygame.KEYDOWN:
            if event.key == pygame.K_RIGHT and self.current_week < len(self.entries) - 1:
                self.current_week += 1
            elif event.key == pygame.K_LEFT and self.current_week > 0:
                self.current_week -= 1
    
    def draw(self):
        """Render journal entry"""
        entry = self.entries[self.current_week]
        
        # Draw background
        self.draw_journal_background()
        
        # Draw period header
        render_text(self.screen, entry["period"], 28, bold=True, y=50)
        
        # Draw summary lines
        y = 120
        for line in entry["lines"]:
            render_text(self.screen, line, 24, y=y, line_height=35)
            y += 35
        
        # Draw practice recommendation if exists
        if entry["needs_practice"]:
            y += 30
            render_text(self.screen, entry["needs_practice"]["title"], 24, bold=True, y=y)
            y += 30
            render_text(self.screen, entry["needs_practice"]["text"], 22, y=y)
        
        # Draw navigation
        self.draw_navigation_controls()
```

## Notes & Considerations

- **Reading Level**: Use tools like Flesch-Kincaid to verify 3rd grade level
- **Tone Testing**: User test with target age group to ensure language resonates
- **Avoid**: Never use discouraging language like "failed", "wrong", "poor"
- **Use Instead**: "Keep practicing", "learning takes time", "every attempt helps"
- **Edge Case**: New student with no data - show welcome message and tutorial
